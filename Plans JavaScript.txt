          features: ["Unlimited scribe-notes", "EHR integration", "Admin dashboard", "Dedicated support", "Custom onboarding & SLA"]
        }
      ]
    }
  };

  ['individual', 'team'].forEach(type => {
    plans[type].yearly = plans[type].monthly.map(p => ({
      ...p,
      price: p.yearly || p.price
    }));
  });

  const planButtons = document.querySelectorAll('.pricing-toggle-tab');
  const billingButtons = document.querySelectorAll('.pricing-billing-tab');
  const billingSwitch = document.getElementById('pricing-billing-switch');
  const plansContainer = document.getElementById('pricing-plans-container');

  let selectedType = 'individual';
  let selectedBilling = 'monthly';

  function renderPlans() {
    const currentPlans = plans[selectedType]?.[selectedBilling];
    if (!currentPlans) return;
    plansContainer.innerHTML = '';
    currentPlans.forEach(plan => {
      const card = document.createElement('div');
      card.className = 'pricing-plan-card';
      card.innerHTML = `
        <h3>${plan.title}</h3>
       
        <p class="pricing-price">${plan.price}</p>
        <ul>${plan.features.map(f => `<li>${f}</li>`).join('')}</ul>
        <a href="${plan.link}" class="pricing-cta ${selectedType}">Choose Plan</a>
      `;
      plansContainer.appendChild(card);
    });
  }

  planButtons.forEach(btn => {
    btn.addEventListener('click', () => {
      planButtons.forEach(b => b.classList.remove('active'));
      btn.classList.add('active');
      selectedType = btn.getAttribute('data-type');
      billingSwitch.style.display = selectedType === 'custom' ? 'none' : 'flex';
      selectedBilling = 'monthly';
      billingButtons.forEach(b => b.classList.remove('active'));
      billingButtons[0].classList.add('active');
      renderPlans();
    });
  });

  billingButtons.forEach(btn => {
    btn.addEventListener('click', () => {
      billingButtons.forEach(b => b.classList.remove('active'));
      btn.classList.add('active');
      selectedBilling = btn.getAttribute('data-billing');
      renderPlans();
    });
  });

  renderPlans();
